# Политический сезон 2023: сдвиги популярности партий, масштабируемые переменными мода (PROPOSAL_spirits.md, пакет 1)
# Значения берутся из переменных UKR_sm_<дух>_<партия>, их считают эффекты UKR_season_* (common/scripted_effects/UKR_season_effects.txt).
# Модификатор живёт, пока у страны есть соответствующий дух; после смены переменных нужен force_update_dynamic_modifier = yes.
# Партии: social_liberal - «Слуга народа», market_liberal - «За будущее», conservative - «Европейская Солидарность»,
# social_democrat - Радикальная партия / «Левый фронт», libertarian_socialist - «Союз левых сил», national_socialist - «Свобода», fascist - «Национальный корпус».

# Кризис доверия к Западу (ukraine_politics.101)
UKR_west_trust_crisis_dynamic = {
	icon = chi_war_of_resistance
	enable = { has_idea = UKR_west_trust_crisis }
	remove_trigger = { NOT = { has_idea = UKR_west_trust_crisis } }

	social_liberal_drift = UKR_sm_west_sl
	market_liberal_drift = UKR_sm_west_ml
	conservative_drift = UKR_sm_west_con
}
