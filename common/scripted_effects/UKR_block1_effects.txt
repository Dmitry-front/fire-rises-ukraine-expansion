# Хроника блока 1 (2020-2022): общие эффекты (DESIGN.md, 4.5)

# Кабинет Гончарука меняется на кабинет Шмыгаля (4 марта 2020 в реальности): премьер и министр обороны
UKR_replace_honcharuk_cabinet = {
	if = {
		limit = { has_idea = UKR_honcharuk }
		remove_ideas = UKR_honcharuk
		add_ideas = UKR_shmyhal
	}
	if = {
		limit = { has_idea = UKR_zagorodnuk }
		remove_ideas = UKR_zagorodnuk
		add_ideas = UKR_taran
	}
}

# Пауза между событиями хроники: 10 дней (см. UKR_chronicle_quiet в common/scripted_triggers/UKR_chronicle_triggers.txt)
UKR_chronicle_cooldown_start = {
	set_country_flag = { flag = UKR_chronicle_cooldown days = 10 }
}
