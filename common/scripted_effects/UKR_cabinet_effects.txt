# Самостоятельный актор кабинета: технократ-министр или член второй линии.
# Перед вызовом: set_temp_variable = { var = UKR_actors_to_add value = N }
# Каждый актор снижает прирост политсилы на 1% (динамический модификатор UKR_cabinet_actors_dynamic)
UKR_add_cabinet_actors = {
	add_to_variable = { var = UKR_cabinet_actors value = UKR_actors_to_add }
	set_variable = { var = UKR_cabinet_actors_pp value = UKR_cabinet_actors }
	multiply_variable = { var = UKR_cabinet_actors_pp value = -0.01 }
	add_dynamic_modifier = { modifier = UKR_cabinet_actors_dynamic }
	force_update_dynamic_modifier = yes
}
