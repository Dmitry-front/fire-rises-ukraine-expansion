# Уровень ПВО 0-3 (UKR_air_defense_level) из очков UKR_air_defense_points (их дают запросы к Германии, Франции и Литве, этап 3)
# и духа приоритета ПВО (решение UKR_air_defense_priority): очки 1 / 3 / 4 дают уровень 1 / 2 / 3, дух добавляет +1, потолок 3.
# Читает ukrainewar.300 (сила и частота ударов по инфраструктуре). Вызывается из on_daily (TFR_on_actions_UKR.txt) во время войны.
# Идея уровня меняется только при смене уровня; разъяснение (ukraine_coalition.70) показывается один раз на уровень 2 и на уровень 3.
UKR_update_air_defense_level = {
	clamp_variable = { var = UKR_air_defense_points min = 0 max = 12 }

	set_temp_variable = { var = UKR_ad_new value = 0 }
	if = {
		limit = { check_variable = { var = UKR_air_defense_points value = 1 compare = greater_than_or_equals } }
		set_temp_variable = { var = UKR_ad_new value = 1 }
	}
	if = {
		limit = { check_variable = { var = UKR_air_defense_points value = 3 compare = greater_than_or_equals } }
		set_temp_variable = { var = UKR_ad_new value = 2 }
	}
	if = {
		limit = { check_variable = { var = UKR_air_defense_points value = 4 compare = greater_than_or_equals } }
		set_temp_variable = { var = UKR_ad_new value = 3 }
	}
	if = {
		limit = { has_idea = UKR_idea_air_defense_priority }
		add_to_temp_variable = { var = UKR_ad_new value = 1 }
		clamp_temp_variable = { var = UKR_ad_new min = 0 max = 3 }
	}

	if = {
		limit = {
			OR = {
				NOT = { check_variable = { var = UKR_air_defense_level value = UKR_ad_new compare = equals } }
				# Страховка: уровень посчитан, а духа нет (старое сохранение, снятие идей)
				AND = {
					check_variable = { var = UKR_ad_new value = 1 compare = equals }
					NOT = { has_idea = UKR_air_defense_level_1 }
				}
				AND = {
					check_variable = { var = UKR_ad_new value = 2 compare = equals }
					NOT = { has_idea = UKR_air_defense_level_2 }
				}
				AND = {
					check_variable = { var = UKR_ad_new value = 3 compare = equals }
					NOT = { has_idea = UKR_air_defense_level_3 }
				}
			}
		}

		# Разъяснение: один раз на уровень 2 и один раз на уровень 3
		if = {
			limit = {
				check_variable = { var = UKR_ad_new value = 3 compare = equals }
				NOT = { has_country_flag = UKR_ad_notice_3 }
			}
			set_country_flag = UKR_ad_notice_3
			set_country_flag = UKR_ad_notice_2
			country_event = { id = ukraine_coalition.70 days = 2 }
		}
		else_if = {
			limit = {
				check_variable = { var = UKR_ad_new value = 2 compare = equals }
				NOT = { has_country_flag = UKR_ad_notice_2 }
			}
			set_country_flag = UKR_ad_notice_2
			country_event = { id = ukraine_coalition.70 days = 2 }
		}

		if = {
			limit = { has_idea = UKR_air_defense_level_1 }
			remove_ideas = UKR_air_defense_level_1
		}
		if = {
			limit = { has_idea = UKR_air_defense_level_2 }
			remove_ideas = UKR_air_defense_level_2
		}
		if = {
			limit = { has_idea = UKR_air_defense_level_3 }
			remove_ideas = UKR_air_defense_level_3
		}
		if = {
			limit = { check_variable = { var = UKR_ad_new value = 1 compare = equals } }
			add_ideas = UKR_air_defense_level_1
		}
		else_if = {
			limit = { check_variable = { var = UKR_ad_new value = 2 compare = equals } }
			add_ideas = UKR_air_defense_level_2
		}
		else_if = {
			limit = { check_variable = { var = UKR_ad_new value = 3 compare = equals } }
			add_ideas = UKR_air_defense_level_3
		}
		set_variable = { var = UKR_air_defense_level value = UKR_ad_new }
	}
}

# Снять духи уровня ПВО (конец войны, postwar). Переменные остаются.
UKR_clear_air_defense_spirits = {
	if = {
		limit = { has_idea = UKR_air_defense_level_1 }
		remove_ideas = UKR_air_defense_level_1
	}
	if = {
		limit = { has_idea = UKR_air_defense_level_2 }
		remove_ideas = UKR_air_defense_level_2
	}
	if = {
		limit = { has_idea = UKR_air_defense_level_3 }
		remove_ideas = UKR_air_defense_level_3
	}
	set_variable = { var = UKR_air_defense_level value = 0 }
}

# Место Украины в штабе коалиции (UKR_hq_seat 0-2). Вход: временная UKR_seat_new (0, 1 или 2).
# 0 - гость, нас слушают; 1 - участник штаба; 2 - постоянное место за столом. Духи меняются вместе с переменной.
UKR_set_hq_seat = {
	set_variable = { var = UKR_hq_seat value = UKR_seat_new }
	if = {
		limit = { has_idea = UKR_coalition_seat_1 }
		remove_ideas = UKR_coalition_seat_1
	}
	if = {
		limit = { has_idea = UKR_coalition_seat_2 }
		remove_ideas = UKR_coalition_seat_2
	}
	if = {
		limit = { check_variable = { var = UKR_seat_new value = 1 compare = equals } }
		add_ideas = UKR_coalition_seat_1
	}
	else_if = {
		limit = { check_variable = { var = UKR_seat_new value = 2 compare = equals } }
		add_ideas = UKR_coalition_seat_2
	}
}

# Что сделали союзники (информационные события ukraine_coalition.80-88). Раз в день проверяем действия стран НАТО из TFR, которые касаются Украины:
# выполненный фокус или активное решение. Флаг UKR_ally_seen_<действие> защищает от повторов. Файлы TFR не трогаем, только читаем.
# has_completed_focus и has_decision читают область страны (GER, FRA, ITA); решения с days_remove считаются активными, пока идёт срок.
# Вызывается из on_daily (TFR_on_actions_UKR.txt) для Украины.
UKR_check_ally_actions = {
	# Германия: фокус GER_aid_ukraine (помощь промышленности Украины)
	if = {
		limit = {
			NOT = { has_country_flag = UKR_ally_seen_ger_aid }
			country_exists = GER
			GER = { has_completed_focus = GER_aid_ukraine }
		}
		set_country_flag = UKR_ally_seen_ger_aid
		country_event = { id = ukraine_coalition.80 days = 1 }
	}
	# Германия: фокус GER_deliver_excess_equipment_to_ukraine (излишки техники)
	if = {
		limit = {
			NOT = { has_country_flag = UKR_ally_seen_ger_surplus }
			country_exists = GER
			GER = { has_completed_focus = GER_deliver_excess_equipment_to_ukraine }
		}
		set_country_flag = UKR_ally_seen_ger_surplus
		country_event = { id = ukraine_coalition.81 days = 1 }
	}
	# Германия: фокус GER_take_in_ukrainian_refugees (беженцы)

			NOT = { has_country_flag = UKR_ally_seen_ger_refugees }
