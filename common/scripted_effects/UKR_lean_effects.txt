	clamp_variable = { var = UKR_digital_lean min = 0 max = 100 }
	clamp_variable = { var = UKR_left_lean min = 0 max = 100 }
	clamp_variable = { var = UKR_authoritarian_lean min = 0 max = 100 }
	clamp_variable = { var = UKR_nationalist_lean min = 0 max = 100 }
}
