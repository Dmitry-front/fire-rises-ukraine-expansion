# Связка украинских организаций MIO с опорами блока 3 (PROPOSAL_concerns.md, раздел 8.1).
# Файл генерируется скриптом; руками правят значения, тексты и ключи локализации держат в паре.
# 1) UKR_mio_hook_<фокус>: награда фокуса-крючка опоры (funds и размер организаций), вызывается из completion_reward.
# 2) UKR_mio_pillar_window_<имя> / UKR_mio_pillar_window_tt_<имя>: бонус опоры при завершении программы окна
#    «Оборонная промышленность» (remove_effect и подсказка в complete_effect решения UKR_mio_*).
# Все mio:-эффекты обёрнуты в has_dlc = "Arms Against Tyranny": без DLC организаций нет.
# Кланы-владельцы (абстракция мода, не утверждение о реальной собственности): met - Малышева, ремонт танков, артиллерия;
# priv - БПЛА, радиоэлектроника, морские дроны; gas - КрАЗ, Форт; sec - Антонов, Мотор Сич, авиаремонт, верфи, Луч.

UKR_mio_hook_t_drone_army = {
	if = {
		limit = { has_dlc = "Arms Against Tyranny" }
		custom_effect_tooltip = UKR_mio_hook_t_drone_army_tt
		hidden_effect = {
			mio:UKR_drone_cluster_organization = {
				add_mio_funds = 300
				add_mio_size = 1
			}
			mio:UKR_sea_drones_organization = {
				add_mio_funds = 150
			}
		}
	}
}

# Завод имени Малышева: клан met
UKR_mio_pillar_window_malyshev = {
	if = {
		limit = { has_dlc = "Arms Against Tyranny" }
		if = {
			limit = {
				has_country_flag = UKR_pillar_chosen_left
			}
			mio:UKR_malyshev_organization = {
				add_mio_funds = 100
			}
			add_stability = 0.01
		}
		if = {
			limit = {
				has_country_flag = UKR_pillar_chosen_oligarchs
				has_country_flag = UKR_ol_vertical_done
			}
			mio:UKR_malyshev_organization = {
				add_mio_funds = 100
			}
			hidden_effect = {
				set_temp_variable = { UKR_ol_amt = 8 }
				UKR_ol_claim_met = yes
				set_temp_variable = { UKR_ol_sh = 1 }
				UKR_ol_share_met = yes
			}
		}
	}
}
