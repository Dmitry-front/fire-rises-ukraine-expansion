# Маркер «НАТО едино» (ТОЛЬКО ЧТЕНИЕ): считаем из идей NATO_unity_1..5, которые выдаёт TFR (событие usa.103).
# Берём Германию, при её отсутствии Францию, затем Британию; идей нет (до usa.103) - ступень 0.
# У PLD ступеней, скорее всего, нет (TFR выдаёт их несуществующей POL), поэтому Польшу не читаем.
# Идеи единства мы не выдаём и не меняем (решение автора, 09.11.2026). Вызывается из on_daily (TFR_on_actions_UKR.txt).
UKR_update_nato_unity_marker = {
	set_variable = { var = UKR_nato_unity_level value = 0 }
	if = {
		limit = { country_exists = GER GER = { has_idea = NATO_unity_5 } }
		set_variable = { var = UKR_nato_unity_level value = 5 }
	}
	else_if = {
		limit = { country_exists = GER GER = { has_idea = NATO_unity_4 } }
		set_variable = { var = UKR_nato_unity_level value = 4 }
	}
	else_if = {
		limit = { country_exists = GER GER = { has_idea = NATO_unity_3 } }
		set_variable = { var = UKR_nato_unity_level value = 3 }
	}
	else_if = {
		limit = { country_exists = GER GER = { has_idea = NATO_unity_2 } }
		set_variable = { var = UKR_nato_unity_level value = 2 }
	}
	else_if = {
		limit = { country_exists = GER GER = { has_idea = NATO_unity_1 } }
		set_variable = { var = UKR_nato_unity_level value = 1 }
	}
	else_if = {
		limit = { country_exists = FRA FRA = { has_idea = NATO_unity_5 } }
		set_variable = { var = UKR_nato_unity_level value = 5 }
	}
	else_if = {
		limit = { country_exists = FRA FRA = { has_idea = NATO_unity_4 } }
		set_variable = { var = UKR_nato_unity_level value = 4 }
	}
	else_if = {
		limit = { country_exists = FRA FRA = { has_idea = NATO_unity_3 } }
		set_variable = { var = UKR_nato_unity_level value = 3 }
	}
	else_if = {
		limit = { country_exists = FRA FRA = { has_idea = NATO_unity_2 } }
		set_variable = { var = UKR_nato_unity_level value = 2 }
	}
	else_if = {
		limit = { country_exists = FRA FRA = { has_idea = NATO_unity_1 } }
		set_variable = { var = UKR_nato_unity_level value = 1 }
	}
	else_if = {
		limit = { country_exists = ENG ENG = { has_idea = NATO_unity_5 } }
		set_variable = { var = UKR_nato_unity_level value = 5 }
	}
	else_if = {
		limit = { country_exists = ENG ENG = { has_idea = NATO_unity_4 } }
		set_variable = { var = UKR_nato_unity_level value = 4 }
	}
	else_if = {
		limit = { country_exists = ENG ENG = { has_idea = NATO_unity_3 } }
		set_variable = { var = UKR_nato_unity_level value = 3 }
	}
	else_if = {
		limit = { country_exists = ENG ENG = { has_idea = NATO_unity_2 } }
		set_variable = { var = UKR_nato_unity_level value = 2 }
	}
	else_if = {
		limit = { country_exists = ENG ENG = { has_idea = NATO_unity_1 } }
		set_variable = { var = UKR_nato_unity_level value = 1 }
	}

	# Множитель объёма помощи в процентах: единое НАТО даёт больше, расколотое меньше
	set_variable = { var = UKR_nato_aid_mult value = 100 }
	if = {
		limit = { check_variable = { var = UKR_nato_unity_level value = 1 compare = equals } }
		set_variable = { var = UKR_nato_aid_mult value = 60 }
	}
	else_if = {
		limit = { check_variable = { var = UKR_nato_unity_level value = 2 compare = equals } }
		set_variable = { var = UKR_nato_aid_mult value = 80 }
	}
	else_if = {
		limit = { check_variable = { var = UKR_nato_unity_level value = 4 compare = equals } }
		set_variable = { var = UKR_nato_aid_mult value = 115 }
	}
	else_if = {
		limit = { check_variable = { var = UKR_nato_unity_level value = 5 compare = equals } }
		set_variable = { var = UKR_nato_aid_mult value = 130 }
	}
}

# Масштаб объёма помощи по единству НАТО. Вход: временная UKR_aid_base (базовое количество), выход: UKR_aid_amount.
# Страховка: если множитель ещё не посчитан (0), считаем 100.
UKR_nato_aid_scale = {
	if = {
		limit = { check_variable = { var = UKR_nato_aid_mult value = 1 compare = less_than } }
		set_variable = { var = UKR_nato_aid_mult value = 100 }
	}
	set_temp_variable = { var = UKR_aid_amount value = UKR_aid_base }
	multiply_temp_variable = { var = UKR_aid_amount value = UKR_nato_aid_mult }
	divide_temp_variable = { var = UKR_aid_amount value = 100 }
	round_temp_variable = UKR_aid_amount
}
