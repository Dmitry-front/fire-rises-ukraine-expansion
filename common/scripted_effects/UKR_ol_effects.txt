# Путь олигархов и вертикали, блок 3 (PROPOSAL_oligarchs.md; DESIGN.md, 8.3).
# Кланы: met (Металл и энергетика), priv (Приват), gas (Газ и химия), sec (Силовой круг).
# Переменные клана: UKR_cl_<id>_share (сырая доля), UKR_cl_<id>_pct (доля в процентах, считает UKR_ol_calc_shares),
# UKR_cl_<id>_claim (претензии 0-100: что президент должен клану), UKR_cl_<id>_grudge (скрытая обида 0-100, копится от давления).
# UKR_ol_loyalty - «Лояльность кланов» (0-100), считает UKR_ol_calc_loyalty.
# Профиль развития «Витрина и подвал» (раздел 13 предложения): снимки UKR_ol_snap_* при старте пути, разница UKR_ol_d_*,
# скрытая «бумага» UKR_ol_paper (около 40% наград Витрины, реализуется в событиях шага 3).
# Проверки выполняются в момент изменения (в эффекте решения, фокуса, события) и раз в месяц (on_monthly, UKR_on_actions_oligarchs.txt).

UKR_ol_start = {
	set_country_flag = UKR_ol_vertical_done
	UKR_ol_apply_oligarchy = yes
	UKR_ol_init_clans = yes
	UKR_ol_snapshot = yes
	UKR_ol_apply_office = yes
	UKR_ol_update = yes
	country_event = { id = ukraine_politics.600 days = 3 }
}

# «Управляемый олигархат» заменяет текущую ступень лестницы олигархата (лестницу назад не поднимаем)
UKR_ol_apply_oligarchy = {
	if = {
		limit = { has_idea = UKR_government_corruption }
		remove_ideas = UKR_government_corruption
	}
	else_if = {
		limit = { has_idea = UKR_government_corruption_low }
		remove_ideas = UKR_government_corruption_low
	}
	else_if = {
		limit = {
			OR = {
				has_idea = UKR_oligarchs_retreating
				has_idea = UKR_oligarchs_remnants
			}
		}
		remove_ideas = UKR_oligarchs_retreating
		remove_ideas = UKR_oligarchs_remnants
		set_country_flag = UKR_oligarchs_restored
	}
	else = {
		set_country_flag = UKR_oligarchs_restored
	}
	add_ideas = UKR_managed_oligarchy
}

# Доли власти в процентах (видны в окне «Вертикаль»)
UKR_ol_calc_shares = {
	set_temp_variable = { UKR_ol_sum = UKR_cl_met_share }
	add_to_temp_variable = { UKR_ol_sum = UKR_cl_priv_share }
	add_to_temp_variable = { UKR_ol_sum = UKR_cl_gas_share }
	add_to_temp_variable = { UKR_ol_sum = UKR_cl_sec_share }
	set_variable = { UKR_cl_met_pct = UKR_cl_met_share }
	multiply_variable = { var = UKR_cl_met_pct value = 100 }
	divide_variable = { var = UKR_cl_met_pct value = UKR_ol_sum }
	round_variable = UKR_cl_met_pct
	set_variable = { UKR_cl_priv_pct = UKR_cl_priv_share }
	multiply_variable = { var = UKR_cl_priv_pct value = 100 }
	divide_variable = { var = UKR_cl_priv_pct value = UKR_ol_sum }
	round_variable = UKR_cl_priv_pct
	set_variable = { UKR_cl_gas_pct = UKR_cl_gas_share }
	multiply_variable = { var = UKR_cl_gas_pct value = 100 }
	divide_variable = { var = UKR_cl_gas_pct value = UKR_ol_sum }
	round_variable = UKR_cl_gas_pct
	set_variable = { UKR_cl_sec_pct = UKR_cl_sec_share }
	multiply_variable = { var = UKR_cl_sec_pct value = 100 }
	divide_variable = { var = UKR_cl_sec_pct value = UKR_ol_sum }
	round_variable = UKR_cl_sec_pct
}

# Лояльность = 100 - (средневзвешенные претензии) - 0.3 * (максимальные претензии одного клана)
UKR_ol_calc_loyalty = {
	set_temp_variable = { UKR_ol_avg = 0 }
	set_temp_variable = { UKR_ol_t = UKR_cl_met_claim }
	multiply_temp_variable = { UKR_ol_t = UKR_cl_met_pct }
	add_to_temp_variable = { UKR_ol_avg = UKR_ol_t }
	set_temp_variable = { UKR_ol_t = UKR_cl_priv_claim }
	multiply_temp_variable = { UKR_ol_t = UKR_cl_priv_pct }
	add_to_temp_variable = { UKR_ol_avg = UKR_ol_t }
	set_temp_variable = { UKR_ol_t = UKR_cl_gas_claim }
	multiply_temp_variable = { UKR_ol_t = UKR_cl_gas_pct }
	add_to_temp_variable = { UKR_ol_avg = UKR_ol_t }
	set_temp_variable = { UKR_ol_t = UKR_cl_sec_claim }
	multiply_temp_variable = { UKR_ol_t = UKR_cl_sec_pct }
	add_to_temp_variable = { UKR_ol_avg = UKR_ol_t }
	divide_temp_variable = { UKR_ol_avg = 100 }

	set_temp_variable = { UKR_ol_max = UKR_cl_met_claim }
	if = {
		limit = { check_variable = { var = UKR_cl_priv_claim value = UKR_ol_max compare = greater_than } }
		set_temp_variable = { UKR_ol_max = UKR_cl_priv_claim }
	}
	if = {
		limit = { check_variable = { var = UKR_cl_gas_claim value = UKR_ol_max compare = greater_than } }
		set_temp_variable = { UKR_ol_max = UKR_cl_gas_claim }
	}
	if = {
		limit = { check_variable = { var = UKR_cl_sec_claim value = UKR_ol_max compare = greater_than } }
		set_temp_variable = { UKR_ol_max = UKR_cl_sec_claim }
	}
	multiply_temp_variable = { UKR_ol_max = 0.3 }

	set_variable = { UKR_ol_loyalty = 100 }
	multiply_temp_variable = { UKR_ol_avg = -1 }
	multiply_temp_variable = { UKR_ol_max = -1 }
	add_to_variable = { UKR_ol_loyalty = UKR_ol_avg }
	add_to_variable = { UKR_ol_loyalty = UKR_ol_max }
	clamp_variable = { var = UKR_ol_loyalty min = 0 max = 100 }
	round_variable = UKR_ol_loyalty
}

# Подводное течение (раздел 6 предложения): чем сильнее вертикаль и обиды, тем выше скрытый рост ОП-ЗЖ (authoritarian_democrat)
UKR_ol_set_undertow = {
	set_variable = { UKR_ol_vertical_level = 0 }
	if = {
		limit = { has_power_balance = { id = UKR_vertical_clans_balance } }
		if = {
			limit = { power_balance_value = { id = UKR_vertical_clans_balance value > 0.1 } }
			set_variable = { UKR_ol_vertical_level = 1 }
		}
		if = {
			limit = { power_balance_value = { id = UKR_vertical_clans_balance value > 0.5 } }
			set_variable = { UKR_ol_vertical_level = 2 }
		}
		if = {
			limit = { power_balance_value = { id = UKR_vertical_clans_balance value > 0.8 } }
			set_variable = { UKR_ol_vertical_level = 3 }
		}
	}
	set_variable = { UKR_ol_grudge_total = UKR_cl_met_grudge }
	add_to_variable = { UKR_ol_grudge_total = UKR_cl_priv_grudge }
	add_to_variable = { UKR_ol_grudge_total = UKR_cl_gas_grudge }
	add_to_variable = { UKR_ol_grudge_total = UKR_cl_sec_grudge }
	if = {
		limit = { check_variable = { var = UKR_ol_grudge_total value = 60 compare = greater_than_or_equals } }
		add_to_variable = { UKR_ol_vertical_level = 1 }
	}
	clamp_variable = { var = UKR_ol_vertical_level min = 0 max = 3 }
	if = {
		limit = { has_idea = UKR_undertow_1 }
		remove_ideas = UKR_undertow_1
	}
	if = {
		limit = { has_idea = UKR_undertow_2 }
		remove_ideas = UKR_undertow_2
	}
	if = {
		limit = { has_idea = UKR_undertow_3 }
		remove_ideas = UKR_undertow_3
	}
	if = {
		limit = { check_variable = { var = UKR_ol_vertical_level value = 3 compare = greater_than_or_equals } }
		add_ideas = UKR_undertow_3
	}
	else_if = {
		limit = { check_variable = { var = UKR_ol_vertical_level value = 2 compare = greater_than_or_equals } }
		add_ideas = UKR_undertow_2
	}
	else_if = {
		limit = { check_variable = { var = UKR_ol_vertical_level value = 1 compare = greater_than_or_equals } }
		add_ideas = UKR_undertow_1
	}
}

			limit = { has_idea = UKR_office_yermak_shadow }

		}
		if = {
			limit = { has_idea = UKR_office_stefanchuk }
			remove_ideas = UKR_office_stefanchuk
		}
		if = {
			limit = { has_country_flag = UKR_yermak_convicted }
			add_ideas = UKR_office_stefanchuk
		}
		else_if = {
			limit = { has_country_flag = UKR_yermak_deal }
