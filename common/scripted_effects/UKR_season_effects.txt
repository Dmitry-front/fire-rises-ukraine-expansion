# Политический сезон 2023: расчёт силы духов (PROPOSAL_spirits.md, пакет 1)
# Схема: событие даёт дух add_timed_idea, затем вызывает UKR_season_<дух> = yes. Эффект считает множитель UKR_sm_k
# (склонности, честность, сила Левого фронта), записывает переменные UKR_sm_<дух>_<партия> и обновляет динамический
# модификатор из common/dynamic_modifiers/UKR_dynamic_season.txt. Вызывать ПОСЛЕ add_timed_idea.
# Единицы *_drift в TFR не проверены в игре: общая сила всех сдвигов регулируется переменной UKR_season_scale (по умолчанию 1).
# Если дрейф слабый - увеличить (например, set_variable = { UKR_season_scale = 5 } в консоли), если сильный - уменьшить.

# Создаёт регулятор силы, если его ещё нет
UKR_season_scale_init = {
	if = {
		limit = { NOT = { has_variable = UKR_season_scale } }
		set_variable = { var = UKR_season_scale value = 1 }
	}
}

# Завершает расчёт множителя: границы и общий регулятор. Вход: временная переменная UKR_sm_k
UKR_season_k_finish = {
	clamp_temp_variable = { var = UKR_sm_k min = 0.6 max = 1.5 }
	multiply_temp_variable = { var = UKR_sm_k value = UKR_season_scale }
}

# Множитель по честности: k = 1 - 0.15 * UKR_honesty_2023 (замалчивание усиливает удар, расследование смягчает)
UKR_season_k_honesty = {
	set_temp_variable = { var = UKR_sm_h value = UKR_honesty_2023 }
	clamp_temp_variable = { var = UKR_sm_h min = -3 max = 3 }
	multiply_temp_variable = { var = UKR_sm_h value = -0.15 }
	set_temp_variable = { var = UKR_sm_k value = 1 }
	add_to_temp_variable = { var = UKR_sm_k value = UKR_sm_h }
	UKR_season_k_finish = yes
}

# Кризис доверия к Западу: цифровая страна переносит урезание грантов легче
# k = 1 + (20 - UKR_digital_lean) / 200; бьёт по «Слуге народа» и «За будущее», чуть поднимает «Европейскую Солидарность»
UKR_season_west_trust_crisis = {
	UKR_season_scale_init = yes
	set_temp_variable = { var = UKR_sm_k value = 20 }
	subtract_from_temp_variable = { var = UKR_sm_k value = UKR_digital_lean }
	divide_temp_variable = { var = UKR_sm_k value = 200 }
	add_to_temp_variable = { var = UKR_sm_k value = 1 }
	UKR_season_k_finish = yes

	set_variable = { var = UKR_sm_west_sl value = -0.015 }
	multiply_variable = { var = UKR_sm_west_sl value = UKR_sm_k }
	set_variable = { var = UKR_sm_west_ml value = -0.012 }
	multiply_variable = { var = UKR_sm_west_ml value = UKR_sm_k }
	set_variable = { var = UKR_sm_west_con value = 0.004 }
	multiply_variable = { var = UKR_sm_west_con value = UKR_sm_k }

	# Партнёры отвернулись: европейское и американское влияние слегка падает
	hidden_effect = {
		add_to_variable = { European_Influence = -0.01 }
		clamp_variable = { var = European_Influence min = 0 max = 1 }
		add_to_variable = { American_Influence = -0.01 }
		clamp_variable = { var = American_Influence min = 0 max = 1 }
	}
	add_dynamic_modifier = { modifier = UKR_west_trust_crisis_dynamic }
	force_update_dynamic_modifier = yes
}
