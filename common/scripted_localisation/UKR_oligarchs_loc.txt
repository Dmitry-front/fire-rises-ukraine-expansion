# Путь олигархов, окно «Вертикаль»: словесные оценки к цифрам (PROPOSAL_oligarchs.md, 13.7).
# Тренд шкалы TFR с начала пути: по разнице UKR_ol_d_* (считает UKR_ol_calc_profile). Пороги: +0.10, +0.03, -0.03, -0.10.
# Использование в локализации: [UKR_OL_TREND_IND] и т.д., [UKR_OL_LOYALTY_WORD]. Ключи слов - UKR_ol_trend_*, UKR_ol_loyalty_*.

defined_text = {
	name = UKR_OL_TREND_IND
	text = {
		trigger = {
			check_variable = { var = UKR_ol_d_ind value = 0.1 compare = greater_than_or_equals }
		}
		localization_key = UKR_ol_trend_up2
	}
	text = {
		trigger = {
			check_variable = { var = UKR_ol_d_ind value = 0.03 compare = greater_than_or_equals }
		}
		localization_key = UKR_ol_trend_up1
	}
	text = {
		trigger = {
			check_variable = { var = UKR_ol_d_ind value = -0.03 compare = greater_than }
		}
		localization_key = UKR_ol_trend_flat
	}
	text = {
		trigger = {
			check_variable = { var = UKR_ol_d_ind value = -0.1 compare = greater_than }
		}
		localization_key = UKR_ol_trend_down1
	}
	text = {
		trigger = { always = yes }
		localization_key = UKR_ol_trend_down2
	}
}
