# Фаза II идёт и война с Россией не закончилась
UKR_coalition_active = {
	has_country_flag = UKR_coalition_phase
	has_war_with = SOV
}

# То же, но первые 120 дней после старта прошли (UKR_coalition_early снимается сам): поздние события ждут
UKR_coalition_mature = {
	UKR_coalition_active = yes
	NOT = { has_country_flag = UKR_coalition_early }
}

# Усталость от войны (UKRwarfatigue, 0-100): высокая и критическая
UKR_fatigue_high = {
	check_variable = { var = UKRwarfatigue value = 55 compare = greater_than_or_equals }
}
UKR_fatigue_critical = {
	check_variable = { var = UKRwarfatigue value = 70 compare = greater_than_or_equals }
}
