# Хроника блока 1 (2020-2022): реальные события + твист TFR + развилка с ценой (DESIGN.md, 4.5)
# Пачка 1: 2020 год. Сроки: окно по дате (date >) и короткий mean_time_to_happen,
# чтобы событие приходило само и игрок его не пропустил, независимо от фокусов.
# Склонности скрыты (hidden_effect), UKR_clamp_leans держит 0-100.

# --- Январь 2020: «Плёнки Гончарука» ---
country_event = {
	id = ukraine.303
	title = ukraine.303.t
	desc = ukraine.303.d
	picture = GFX_UKR_big_corruption_scandal

	fire_only_once = yes

	trigger = {
		tag = UKR
		date > 2020.1.14
		has_idea = UKR_honcharuk
		has_country_leader = {
			character = UKR_volodymyr_zelensky
			ruling_only = yes
		}
	}

	mean_time_to_happen = {
		days = 3
	}

	# Принять отставку
	option = {
		name = ukraine.303.a
		add_stability = -0.02
		add_political_power = -30
		set_country_flag = UKR_pm_replaced_early
		UKR_replace_honcharuk_cabinet = yes
		hidden_effect = {
			add_to_variable = { UKR_authoritarian_lean = 1 }
			UKR_clamp_leans = yes
		}
		ai_chance = { factor = 35 }
	}

	# Дать последний шанс
	option = {
		name = ukraine.303.b
		add_timed_idea = { idea = UKR_last_chance_cabinet days = 45 }
		hidden_effect = {
			country_event = { id = ukraine.304 days = 45 }
			add_to_variable = { UKR_digital_lean = 1 }
			UKR_clamp_leans = yes
		}
		ai_chance = { factor = 65 }
	}
}
