country_event = {
	id = ukraine_foreign.300
	title = ukraine_foreign.300.t
	desc = ukraine_foreign.300.d

	option = {
		name = UKR_req_opt_full
		ai_chance = {
			base = 25
			modifier = { add = 25 UKR_ally_opinion_high = yes }
			modifier = { add = 10 UKR_ally_opinion_mid = yes }
			modifier = { add = -20 UKR_ally_opinion_low = yes }
			modifier = { add = 25 UKR_ally_trust_top = yes }
			modifier = { add = 15 UKR_ally_trust_high = yes }
			modifier = { add = -10 UKR_ally_trust_low = yes }
			modifier = { add = -20 UKR_ally_trust_none = yes }
			modifier = { add = 10 has_global_flag = SOV_nato_war_begins_global }
			modifier = { add = -15 UKR = { has_country_flag = UKR_np_hero_upa } }
		}
		add_opinion_modifier = { target = UKR modifier = ukr_allies_request_met }
		UKR = {
			clr_country_flag = UKR_req_pl_hub_pending
			set_country_flag = { flag = UKR_req_pl_hub_full days = 10 }
			if = {
				limit = { has_idea = UKR_pl_hub_idea_lim }
				remove_ideas = UKR_pl_hub_idea_lim

			add_timed_idea = {
				idea = UKR_pl_hub_idea

				type = support_equipment_1
				amount = UKR_aid_amount
				producer = PLD
			}
			set_country_flag = UKR_pl_hub_open
			country_event = { id = ukraine_foreign.320 days = 150 }
			country_event = { id = ukraine_foreign.310 days = 1 }

			modifier = { add = 10 UKR_ally_trust_low = yes }
			modifier = { add = 5 UKR_ally_trust_none = yes }
		}
		add_opinion_modifier = { target = UKR modifier = ukr_allies_request_partial }
		UKR = {
			clr_country_flag = UKR_req_pl_hub_pending
			set_country_flag = { flag = UKR_req_pl_hub_part days = 10 }
			if = {
				limit = { NOT = { has_idea = UKR_pl_hub_idea } }
				add_timed_idea = {
					idea = UKR_pl_hub_idea_lim
					days = 180
				}
			}
			set_temp_variable = { var = UKR_aid_base value = 30 }
			UKR_nato_aid_scale = yes
			add_equipment_to_stockpile = {
