	title = ukraine_politics.720.t
	desc = ukraine_politics.720.d

	trigger = {
		tag = UKR
		date > 2025.8.29
		has_country_flag = UKR_np_convent_done

			set_country_flag = UKR_parubiy_killed

				limit = { has_idea = UKR_parubiy_face }

				add_ideas = UKR_parubiy_legacy

		custom_effect_tooltip = UKR_np_demo_up_tt
		hidden_effect = {
			add_to_variable = { UKR_np_demo = 3 }
