# Путь олигархов и вертикали, блок 3, шаг 1 (PROPOSAL_oligarchs.md, разделы 8, 9, 11; DESIGN.md, 8.3).
# 600: «Обед в Офисе» (запускается через 3 дня после фокуса UKR_ol_vertical эффектом UKR_ol_start).
# 601-605: торг о кабинете (премьер, экономика, МВД, оборона, разведка). Назначение министра от клана погашает его претензии,
# назначение «человека президента» сдвигает шкалу UKR_vertical_clans_balance к вертикали. Отказ не требует политсилы.
# Реальные люди: активы и должности пишем как факты; выдуманных цитат и сумм нет.
# События 610+ (счета кланов, «Строптивый», конституционный пакет, «Итоги года» и др.) добавляются в шагах 2-3.

# Обед в Офисе: старт пути
country_event = {
	id = ukraine_politics.600
	title = ukraine_politics.600.t
	desc = {
		text = ukraine_politics.600.d_stefanchuk
		trigger = {
			has_idea = UKR_office_stefanchuk
		}
	}
	desc = {
		text = ukraine_politics.600.d_yermak
		trigger = {
			NOT = { has_idea = UKR_office_stefanchuk }
		}
	}
	picture = GFX_report_event_ukrainian_civil_war

	is_triggered_only = yes

	trigger = {
		tag = UKR
	}

	# Договариваться
	option = {
		name = ukraine_politics.600.a
		custom_effect_tooltip = UKR_ol_bop_clans_tt
		custom_effect_tooltip = UKR_ol_claims_all_down_tt
		add_power_balance_value = {
			id = UKR_vertical_clans_balance
			value = -0.10
		}
		hidden_effect = {
			set_temp_variable = { UKR_ol_amt = -5 }
			UKR_ol_claim_met = yes
			UKR_ol_claim_priv = yes
			UKR_ol_claim_gas = yes
			UKR_ol_claim_sec = yes
			add_to_variable = { UKR_authoritarian_lean = 1 }
			UKR_clamp_leans = yes
			country_event = { id = ukraine_politics.601 days = 2 }
		}
		ai_chance = { factor = 30 }
	}

	# Держать дистанцию
	option = {
		name = ukraine_politics.600.b
		hidden_effect = {
			country_event = { id = ukraine_politics.601 days = 2 }
		}
		ai_chance = { factor = 40 }
	}

	# Напомнить, чья Банковая
	option = {
		name = ukraine_politics.600.c
		custom_effect_tooltip = UKR_ol_bop_vertical_tt
		custom_effect_tooltip = UKR_ol_grudge_up_tt
		add_power_balance_value = {
			id = UKR_vertical_clans_balance
			value = 0.10
		}
		hidden_effect = {
			add_to_variable = { UKR_cl_met_grudge = 5 }
			add_to_variable = { UKR_cl_priv_grudge = 5 }
			add_to_variable = { UKR_cl_gas_grudge = 5 }
			add_to_variable = { UKR_cl_sec_grudge = 5 }
			UKR_ol_update = yes
			add_to_variable = { UKR_authoritarian_lean = 2 }
			UKR_clamp_leans = yes
			country_event = { id = ukraine_politics.601 days = 2 }
		}
		ai_chance = { factor = 30 }
	}
}
