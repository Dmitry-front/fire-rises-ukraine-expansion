	id = ukraine_politics.600
	title = ukraine_politics.600.t
	desc = {
		text = ukraine_politics.600.d_stefanchuk
		trigger = {
			has_idea = UKR_office_stefanchuk
		}
	}
	desc = {
		text = ukraine_politics.600.d_yermak
		trigger = {
			NOT = { has_idea = UKR_office_stefanchuk }
		}
	}
	picture = GFX_report_event_ukrainian_civil_war

	is_triggered_only = yes

	trigger = {
		tag = UKR
	}

	# Договариваться
	option = {
		name = ukraine_politics.600.a
		custom_effect_tooltip = UKR_ol_bop_clans_tt
		custom_effect_tooltip = UKR_ol_claims_all_down_tt
		add_power_balance_value = {
			id = UKR_vertical_clans_balance
			value = -0.10
		}
		hidden_effect = {
			set_temp_variable = { UKR_ol_amt = -5 }
			UKR_ol_claim_met = yes
			UKR_ol_claim_priv = yes
			UKR_ol_claim_gas = yes
			UKR_ol_claim_sec = yes
			add_to_variable = { UKR_authoritarian_lean = 1 }
			UKR_clamp_leans = yes
			country_event = { id = ukraine_politics.601 days = 2 }
		}
		ai_chance = { factor = 30 }
	}

	# Держать дистанцию
	option = {
		name = ukraine_politics.600.b
		hidden_effect = {
			country_event = { id = ukraine_politics.601 days = 2 }
		}
		ai_chance = { factor = 40 }
	}

	# Напомнить, чья Банковая
	option = {
		name = ukraine_politics.600.c
		custom_effect_tooltip = UKR_ol_bop_vertical_tt
		custom_effect_tooltip = UKR_ol_grudge_up_tt

			id = UKR_vertical_clans_balance
			value = 0.10
		}
		hidden_effect = {
			add_to_variable = { UKR_cl_met_grudge = 5 }
			add_to_variable = { UKR_cl_priv_grudge = 5 }
			add_to_variable = { UKR_cl_gas_grudge = 5 }
			add_to_variable = { UKR_cl_sec_grudge = 5 }

	id = ukraine_politics.616
	title = ukraine_politics.616.t
	desc = ukraine_politics.616.d
	picture = GFX_report_event_ukrainian_civil_war

	trigger = {
		tag = UKR
		has_country_flag = UKR_ol_vertical_done
		has_country_flag = UKR_ol_cabinet_formed
		NOT = { has_country_flag = UKR_ol_target_cd }
		NOT = { has_country_flag = UKR_ol_target_met }
		OR = {
			check_variable = { var = UKR_cl_met_claim value = 45 compare = greater_than_or_equals }
			check_variable = { var = UKR_cl_met_grudge value = 25 compare = greater_than_or_equals }
		}
	}

	mean_time_to_happen = {
		days = 45
	}

	immediate = {
		set_country_flag = {
			flag = UKR_ol_target_cd
			days = 45
		}
	}

	# Открыть дело: клан становится целью давления
	option = {
		name = ukraine_politics.616.a
		add_political_power = 10
		custom_effect_tooltip = UKR_ol_target_met_tt
		hidden_effect = {
			set_country_flag = UKR_ol_target_met
			add_to_variable = { UKR_authoritarian_lean = 1 }
			UKR_clamp_leans = yes
		}
		ai_chance = { factor = 35 }
	}

	# Уступить
	option = {
		name = ukraine_politics.616.b
		trigger = { has_political_power > 14 }
		add_political_power = -15
		custom_effect_tooltip = UKR_ol_bop_clans_tt
