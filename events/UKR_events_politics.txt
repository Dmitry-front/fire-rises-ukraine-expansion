	fire_only_once = yes

	trigger = {
		tag = UKR
		date > 2022.12.31
		NOT = { has_country_flag = UKR_politics_2023_started }
		has_country_leader = {
			character = UKR_volodymyr_zelensky
			ruling_only = yes

	}

		set_country_flag = UKR_politics_2023_started
		set_variable = { UKR_honesty_2023 = 0 }
